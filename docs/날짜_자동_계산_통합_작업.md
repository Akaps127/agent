```python
def format_date_str(date_str):
    if not date_str: return ""
    try:
        dt = datetime.strptime(date_str, "%Y-%m-%d %H:%M")
        return dt.strftime("%Y. %m. %d.(%H:%M)")
    except:
        return date_str
```

```typescript
// 영업일 계산 (주말 제외)
function addBusinessDays(startDate: dayjs.Dayjs, days: number): dayjs.Dayjs {
  let current = startDate;
  let added = 0;
  
  while (added < days) {
    current = current.add(1, 'day');
    if (current.day() !== 0 && current.day() !== 6) {
      added++;
    }
  }
  
  return current;
}

// 낙찰 방법에 따른 날짜 계산
function calculateBidDates(winningMethod: string) {
  const today = dayjs();
  
  if (winningMethod === "소액수의") {
    // 소액수의: 3 영업일 (주말 제외)
    const openingDate = addBusinessDays(today, 3);
    return {
      submission_start: today.hour(9).minute(0).format('YYYY-MM-DD HH:mm'),
      submission_end: openingDate.hour(10).minute(0).format('YYYY-MM-DD HH:mm'),
      opening_datetime: openingDate.hour(11).minute(0).format('YYYY-MM-DD HH:mm'),
    };
  } else {
    // 적격심사: 7 달력일 (주말 포함)
    const openingDate = today.add(7, 'day');
    return {
      submission_start: today.hour(9).minute(0).format('YYYY-MM-DD HH:mm'),
      submission_end: openingDate.hour(10).minute(0).format('YYYY-MM-DD HH:mm'),
      opening_datetime: openingDate.hour(11).minute(0).format('YYYY-MM-DD HH:mm'),
    };
  }
}
```

## 📋 기술 스펙

### 날짜 계산 규칙
- **소액수의**: 오늘 + 3 영업일 (주말 제외)
- **적격심사**: 오늘 + 7 달력일 (주말 포함)

### 시간 설정
- **제출 시작**: 09:00
- **제출 마감**: 10:00
- **개찰**: 11:00

### 날짜 포맷
- **Frontend 입력**: `YYYY-MM-DD HH:mm` (예: `2025-12-25 09:00`)
- **Backend 출력**: `YYYY. MM. DD.(HH:MM)` (예: `2025. 12. 25.(09:00)`)
